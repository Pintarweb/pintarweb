let allLeads = [];
let allHunts = [];
let currentFolder = 'active';
let sortMode = 'date'; // Primary Vector: Newest First

function switchView(view) {
    const leadsView = document.getElementById('leads-view');
    const huntsView = document.getElementById('hunts-view');
    const navLeads = document.getElementById('nav-leads');
    const navHunts = document.getElementById('nav-hunts');

    if (view === 'hunts') {
        leadsView.classList.add('hidden');
        huntsView.classList.remove('hidden');
        navHunts.className = "px-5 py-2 rounded-lg text-xs font-black transition-all bg-slate-900 text-white shadow-sm";
        navLeads.className = "px-5 py-2 rounded-lg text-xs font-black transition-all text-slate-500";
        fetchHunts();
    } else {
        leadsView.classList.remove('hidden');
        huntsView.classList.add('hidden');
        navLeads.className = "px-5 py-2 rounded-lg text-xs font-black transition-all bg-slate-900 text-white shadow-sm";
        navHunts.className = "px-5 py-2 rounded-lg text-xs font-black transition-all text-slate-500";
    }
}

function updateHuntCommand() {
    const cat = document.getElementById('hunt-category').value;
    const loc = document.getElementById('hunt-location').value;
    const lim = document.getElementById('hunt-limit').value;
    const sources = Array.from(document.querySelectorAll('.source-check:checked')).map(el => el.value).join(',');
    const cmd = `npx tsx src/index.ts --category "${cat}" --location "${loc}" --limit ${lim} --sources ${sources}`;
    document.getElementById('command-output').innerText = cmd;
}

function copyCommand(e) {
    const cmd = document.getElementById('command-output').innerText;
    navigator.clipboard.writeText(cmd);
    const btn = (e && e.currentTarget) || (window.event && window.event.target) || event.target;
    if (!btn) return;
    const original = btn.innerText;
    btn.innerText = "COPIED! ✅";
    setTimeout(() => btn.innerText = original, 2000);
}

async function fetchHunts() {
    try {
        const res = await fetch('/api/hunts?t=' + Date.now());
        allHunts = await res.json();
        populateSmartLists();
        renderHunts();
    } catch (e) { console.error("Hunt history fetch failure", e); }
}

function populateSmartLists() {
    const catList = document.getElementById('category-list');
    const locList = document.getElementById('location-list');

    if (!catList || !locList) return;

    const cats = new Set();
    const locs = new Set();

    // Harvesting Intelligence
    if (Array.isArray(allHunts)) {
        allHunts.forEach(h => {
            if (h.category) cats.add(h.category);
            if (h.location) locs.add(h.location);
        });
    }

    if (Array.isArray(allLeads)) {
        allLeads.forEach(l => {
            if (l.category) cats.add(l.category);
        });
    }

    console.log(`[SmartList] Synced ${cats.size} sectors and ${locs.size} regions.`);

    // UI Feedback: Update badges
    const catBadge = document.getElementById('cat-sync-badge');
    const locBadge = document.getElementById('loc-sync-badge');
    if (catBadge) catBadge.innerText = `${cats.size} KNOWN`;
    if (locBadge) locBadge.innerText = `${locs.size} KNOWN`;

    catList.innerHTML = Array.from(cats).filter(Boolean).sort().map(c => `<option value="${c}">`).join('');
    locList.innerHTML = Array.from(locs).filter(Boolean).sort().map(l => `<option value="${l}">`).join('');
}

function renderHunts() {
    const container = document.getElementById('hunts-container');
    if (!container) return;
    if (!allHunts || allHunts.length === 0) {
        container.innerHTML = '<tr><td colspan="4" class="py-10 text-center text-slate-400 font-bold uppercase tracking-widest">No history recorded</td></tr>';
        return;
    }
    container.innerHTML = allHunts.map(h => `
        <tr class="border-b border-slate-50 hover:bg-slate-50 transition-colors">
            <td class="py-4 font-bold text-slate-400 text-xs">${new Date(h.created_at).toLocaleDateString()}</td>
            <td class="py-4">
                <p class="font-black text-slate-900 text-sm">${h.category}</p>
                <p class="text-[10px] text-blue-600 font-black uppercase tracking-tighter">${h.location}</p>
            </td>
            <td class="py-4">
                <div class="flex gap-1">
                    ${(h.sources || '').split(',').map(s => `<span class="bg-slate-100 text-[9px] font-black px-1.5 py-0.5 rounded uppercase">${s}</span>`).join('')}
                </div>
            </td>
            <td class="py-4 font-black text-slate-900 text-base">${h.leads_found || 0} <span class="text-[9px] text-slate-400 uppercase tracking-tighter">Leads</span></td>
        </tr>
    `).join('');
}

function openModal(contentHtml) {
    const modal = document.getElementById('scorecard-modal');
    document.getElementById('modal-content').innerHTML = contentHtml;
    modal.classList.remove('hidden');
    setTimeout(() => {
        modal.classList.add('opacity-100');
        modal.firstElementChild.classList.add('scale-100');
    }, 10);
}

function closeModal() {
    const modal = document.getElementById('scorecard-modal');
    modal.classList.remove('opacity-100');
    modal.firstElementChild.classList.remove('scale-100');
    setTimeout(() => modal.classList.add('hidden'), 300);
}

async function updateStatus(phone, status) {
    try {
        await fetch('/api/leads', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ phone_normalized: phone, status }) });
        fetchLeads();
    } catch (e) { console.error(e); }
}

async function deleteLead(phone) {
    if (!confirm('Confirm permanent deletion?')) return;
    try {
        await fetch('/api/leads', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ phone_normalized: phone }) });
        fetchLeads();
    } catch (e) { console.error(e); }
}

}

function exportToCRM() {
    const engagedLeads = allLeads.filter(l => l.status === 'Contacted' || l.status === 'Demo Built');
    
    if (engagedLeads.length === 0) {
        alert("No contacted leads available to export.");
        return;
    }

    const headers = ["Business Name", "Phone", "Category", "Location", "Status", "Score", "Website"];
    
    const csvRows = [];
    csvRows.push(headers.join(','));
    
    for (const lead of engagedLeads) {
        const values = [
            `"${(lead.business_name || '').replace(/"/g, '""')}"`,
            `"${lead.phone_normalized || ''}"`,
            `"${lead.category || ''}"`,
            `"${lead.location || ''}"`,
            `"${lead.status || ''}"`,
            `${lead.lead_score || 0}`,
            `"${lead.website_url || ''}"`
        ];
        csvRows.push(values.join(','));
    }
    
    const csvString = csvRows.join('\n');
    const blob = new Blob([csvString], { type: 'text/csv;charset=utf-8;' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.style.display = 'none';
    a.href = url;
    a.download = `PintarWeb_Contacted_Leads_${new Date().toISOString().split('T')[0]}.csv`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
}

function renderLeads() {
    const container = document.getElementById('leads-container');
    const categoryEl = document.getElementById('filter-category');
    const statusEl = document.getElementById('filter-status');
    const websiteEl = document.getElementById('filter-website');
    const sourceEl = document.getElementById('filter-source');
    const arrivalEl = document.getElementById('filter-arrival');
    const scoreEl = document.getElementById('filter-score');
    
    if (!container || !categoryEl || !statusEl) {
        console.error("Critical DOM elements missing for renderLeads");
        return;
    }

    const category = categoryEl.value;
    const status = statusEl.value;
    const website = websiteEl.value;
    const source = sourceEl.value;
    const arrival = arrivalEl.value;
    const minScore = parseInt(scoreEl.value || 0);
    const search = (document.getElementById('search').value || '').toLowerCase();
    const now = new Date();

    // Tactical Intelligence: Find the start time of the most recent mission
    const latestHuntTime = (allHunts && allHunts.length > 0) ? new Date(allHunts[0].created_at + 'Z').getTime() : 0;

    let filtered = allLeads.filter(l => {
        const isArchived = l.status === 'Archived';
        if (currentFolder === 'active' && isArchived) return false;
        if (currentFolder === 'archived' && !isArchived) return false;

        // Using updated_at for the "Arrival" filters
        const leadDate = new Date((l.updated_at || l.created_at) + 'Z');
        const age = now - leadDate;
        if (arrival === 'Today' && age > 86400000) return false;
        if (arrival === 'Week' && age > 604800000) return false;

        if (category !== 'All' && l.category !== category) return false;
        if (status === 'Engaged') {
            if (l.status !== 'Contacted' && l.status !== 'Demo Built') return false;
        } else if (status !== 'All' && l.status !== status) {
            return false;
        }
        if (website === 'Yes' && !l.website_url) return false;
        if (website === 'No' && l.website_url) return false;
        if (source !== 'All' && !(l.source_origin || '').includes(source)) return false;
        if ( (l.lead_score || 0) < minScore) return false;
        if (!l.business_name.toLowerCase().includes(search)) return false;
        return true;
    });

    // Primary Sort: Updated At (Interaction Time) - with safety
    filtered.sort((a, b) => {
        if (sortMode === 'score') return (b.lead_score || 0) - (a.lead_score || 0);
        const dateA = new Date((a.updated_at || a.created_at) + 'Z').getTime();
        const dateB = new Date((b.updated_at || b.created_at) + 'Z').getTime();
        return (dateB || 0) - (dateA || 0);
    });

    // Global Census Counter
    const activeCount = allLeads.filter(l => l.status !== 'Archived').length;
    const totalCount = allLeads.length;
    const countEl = document.getElementById('lead-count');
    if (countEl) countEl.innerText = `${activeCount} ACTIVE / ${totalCount} TOTAL`;

    const sortBtn = document.getElementById('sort-btn');
    if (sortBtn) sortBtn.innerText = sortMode === 'score' ? 'Highest Score ↓' : 'Newest Arrival ↓';

    if (filtered.length === 0) {
        container.innerHTML = '<div class="text-center py-24 bg-white rounded-2xl border border-slate-200 text-slate-400 font-bold uppercase tracking-widest">No matching leads in this sector</div>';
        return;
    }

    container.innerHTML = filtered.map(lead => {
        const isArchived = lead.status === 'Archived';
        const leadUpdateTime = new Date((lead.updated_at || lead.created_at) + 'Z').getTime();

        // Surgical Tagging: Only pulse if update happened within 15 min of the hunt log
        const isFreshFromMission = latestHuntTime > 0 && Math.abs(leadUpdateTime - latestHuntTime) < 900000;

        const actions = !isArchived ? `
            <button onclick="updateStatus('${lead.phone_normalized}', 'Demo Built')" class="text-[10px] font-black text-slate-400 hover:text-blue-600 uppercase tracking-widest transition-colors">Mark Demo</button>
            <button onclick="updateStatus('${lead.phone_normalized}', 'Contacted')" class="text-[10px] font-black text-slate-400 hover:text-orange-600 uppercase tracking-widest transition-colors">Mark Contacted</button>
            <button onclick="updateStatus('${lead.phone_normalized}', 'Archived')" class="text-[10px] font-black text-slate-400 hover:text-red-600 uppercase tracking-widest ml-auto transition-colors">Archive 📁</button>
        ` : `
            <button onclick="updateStatus('${lead.phone_normalized}', 'New')" class="text-[10px] font-black text-blue-600 uppercase tracking-widest transition-colors">Reinstate 🔄</button>
            <button onclick="deleteLead('${lead.phone_normalized}')" class="text-[10px] font-black text-red-600 uppercase tracking-widest ml-auto transition-colors">Delete ❌</button>
        `;

        return `
        <div class="group bg-white p-6 rounded-3xl border border-slate-100 shadow-sm hover:shadow-xl hover:border-blue-200 transition-all duration-300 relative ${isFreshFromMission ? 'ring-4 ring-blue-600 ring-offset-2 shadow-2xl z-10' : ''}">
            ${isFreshFromMission ? `

async function fetchLeads() {
    try {
        const res = await fetch('/api/leads?t=' + Date.now());
        if (!res.ok) throw new Error("Failed to fetch leads: " + res.status);
        allLeads = await res.json();
        if (!Array.isArray(allLeads)) throw new Error("Invalid leads data received");

        const catSelect = document.getElementById('filter-category');
        if (!catSelect) return;
        const activeLeads = allLeads.filter(l => l.status !== 'Archived');
        const stats = activeLeads.reduce((acc, l) => {
            if (l.category) acc[l.category] = (acc[l.category] || 0) + 1;
            return acc;
        }, {});

        const cats = ['All', ...Object.keys(stats).filter(s => s && s !== 'null').sort()];
        catSelect.innerHTML = cats.map(c => `<option value="${c}">${c === 'All' ? 'All Industries (' + activeLeads.length + ')' : c + ' (' + stats[c] + ')'}</option>`).join('');

        if (!catSelect.value) catSelect.value = 'All';

        populateSmartLists();
        renderLeads();
    } catch (e) { 
        console.error("Leads fetch failure", e);
        const container = document.getElementById('leads-container');
        if (container) container.innerHTML = `<div class="text-center py-24 text-red-500 font-black">CRITICAL FAILURE: ${e.message}</div>`;
    }
}

document.addEventListener('DOMContentLoaded', () => {
    // Global Error Catcher for easier debugging on screen
    window.onerror = function(msg, url, line, col, error) {
        const container = document.getElementById('leads-container');
        if (container) {
            container.innerHTML = `<div class="p-6 bg-red-50 border-2 border-red-500 rounded-3xl text-red-600">
                <p class="font-black uppercase mb-2">Internal Script Error</p>
                <p class="text-xs font-mono">${msg}</p>
                <p class="text-[10px] mt-2 opacity-50">${url}:${line}:${col}</p>
            </div>`;
        }
        return false;
    };

    const searchEl = document.getElementById('search');
    if (searchEl) {
        searchEl.addEventListener('input', applyFilters);
    }
    fetchLeads();
    fetchHunts();
    setInterval(fetchLeads, 30000);
    setInterval(fetchHunts, 30000);
});
